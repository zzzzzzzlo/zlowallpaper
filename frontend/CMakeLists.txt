cmake_minimum_required(VERSION 3.21)
project(ZloWallpaper VERSION 0.2.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Multimedia MultimediaWidgets Network Concurrent)
set(ZLO_RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/deps" CACHE PATH "Local external runtime packages")
set(MPV_ROOT "${ZLO_RUNTIME_ROOT}/mpv/package")
if(NOT EXISTS "${MPV_ROOT}/include/mpv/client.h")
  message(FATAL_ERROR "Missing libmpv development package. See docs/FRONTEND.md")
endif()
file(GLOB_RECURSE ZLO_SOURCES CONFIGURE_DEPENDS src/*.cpp src/*.h)
list(FILTER ZLO_SOURCES EXCLUDE REGEX "/tests/")
add_executable(ZloWallpaper WIN32 ${ZLO_SOURCES} resources/asterol.rc resources/resources.qrc)
target_include_directories(ZloWallpaper PRIVATE src "${MPV_ROOT}/include")
target_link_libraries(ZloWallpaper PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Network Qt6::Concurrent user32 shell32 advapi32 ole32 dxgi crypt32 "${MPV_ROOT}/libmpv.dll.a")
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}" ZLO_SOURCE_ROOT)
target_compile_definitions(ZloWallpaper PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN LIGHTWALLPAPER_SOURCE_ROOT="${ZLO_SOURCE_ROOT}")
foreach(pair "${MPV_ROOT}/libmpv-2.dll" "${ZLO_RUNTIME_ROOT}/ffmpeg/package/ffmpeg.exe")
  if(NOT EXISTS "${pair}")
    message(FATAL_ERROR "Missing runtime: ${pair}")
  endif()
  add_custom_command(TARGET ZloWallpaper POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${pair}" "$<TARGET_FILE_DIR:ZloWallpaper>/")
endforeach()
foreach(helper icontra translucenttb)
  if(EXISTS "${ZLO_RUNTIME_ROOT}/${helper}")
    add_custom_command(TARGET ZloWallpaper POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${ZLO_RUNTIME_ROOT}/${helper}" "$<TARGET_FILE_DIR:ZloWallpaper>/${helper}")
  else()
    message(FATAL_ERROR "Missing ${helper} runtime. All desktop features require this package.")
  endif()
endforeach()
option(ZLO_BUILD_TESTS "Build storefront integration tests" ON)
if(ZLO_BUILD_TESTS)
  enable_testing()
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(StoreTests tests/StoreTests.cpp src/ui/StorePage.cpp src/ui/StorePage.h src/store/ApiClient.cpp src/store/ApiClient.h src/store/CredentialVault.cpp src/store/StoreService.cpp src/store/StoreService.h src/store/DownloadManager.cpp src/store/DownloadManager.h src/store/DemoServer.cpp src/store/DemoServer.h src/library/WallpaperLibrary.cpp src/library/WallpaperLibrary.h src/library/WallpaperEntry.cpp src/media/ThumbnailGenerator.cpp)
  target_include_directories(StoreTests PRIVATE src)
  target_compile_definitions(StoreTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(StoreTests PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Test Qt6::Concurrent crypt32 user32)
  add_test(NAME StoreIntegration COMMAND StoreTests -o -,txt -o integration-result.txt,txt)
  set_tests_properties(StoreIntegration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
